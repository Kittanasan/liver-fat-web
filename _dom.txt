function copyText(t){
  const legacy=()=>new Promise(res=>{
    try{
      const ta=document.createElement("textarea");ta.value=t;ta.setAttribute("readonly","");
      ta.style.position="fixed";ta.style.top="-1000px";ta.style.opacity="0";
      document.body.appendChild(ta);ta.select();ta.setSelectionRange(0,t.length);
      const ok=document.execCommand&&document.execCommand("copy");

    }catch(e){res(false)}
  });
  if(navigator.clipboard&&navigator.clipboard.writeText&&window.isSecureContext)

function iou(a,b){const x1=Math.max(a[0],b[0]),y1=Math.max(a[1],b[1]),x2=Math.min(a[2],b[2]),y2=Math.min(a[3],b[3]);
  const i=Math.max(0,x2-x1)*Math.max(0,y2-y1);return i/((a[2]-a[0])*(a[3]-a[1])+(b[2]-b[0])*(b[3]-b[1])-i+1e-9)}
